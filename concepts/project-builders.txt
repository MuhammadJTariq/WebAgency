 add_action('admin_init', [$this, 'redirect_project']);
        add_action('admin_init', [$this, 'save_project_builder']);
        add_action('admin_menu', [$this, 'project_builder']);



         public function save_project_builder(){
        if(!isset($_POST['project_nonce']) || wp_verify_nonce($_POST['project_nonce'], 'save_project')){
            return;
        }

        $post_id = wp_insert_post([
        'ID'           => $_GET['post_id'] ?? 0,
        'post_type'    => 'project',
        'post_title'   => sanitize_text_field($_POST['title']),
        'post_content' => sanitize_textarea_field($_POST['description']),
        'post_status'  => 'publish',
        ]);

        wp_redirect(admin_url('admin.php?page=project-builder&post_id=' . $post_id));
        exit;
    }

    public function project_builder(){
        add_menu_page(
            'Project Builder',
            'Project Builder',
            'edit_posts',
            'project-builder',
            [$this, 'render_project_builder']
        );
    }

    public function render_project_builder(){
         $post_id = isset($_GET['post_id']) ? intval($_GET['post_id']) : 0;
    $post = $post_id ? get_post($post_id) : null;

    ?>
    <div class="wrap">
        <h1><?php echo $post ? 'Edit Project' : 'New Project'; ?></h1>

        <form method="post">
            <?php wp_nonce_field('save_project', 'project_nonce'); ?>

            <input type="text" name="title"
                   placeholder="Project Title"
                   value="<?php echo esc_attr($post->post_title ?? ''); ?>">

            <textarea name="description"
                      placeholder="Project Description"><?php
                echo esc_textarea($post->post_content ?? '');
            ?></textarea>

            <button type="submit">Save Project</button>
        </form>
    </div>
    <?php

    }

    public function redirect_project(){
        global $pagenow;
        if(!is_admin()){
            return;
        }
        if($pagenow === 'post-new.php' && isset($_GET['post_type']) && $_GET['post_type'] === 'project'){
            wp_redirect( admin_url('admin.php?page=project-builder') );
            exit;
        }
     if ($pagenow === 'post.php' && isset($_GET['post']) && get_post_type($_GET['post']) === 'project') {
        wp_redirect(admin_url('admin.php?page=project-builder&post_id=' . intval($_GET['post'])));
        exit;
    }
    }
